	public static void main(String[] args) {
		Scanner scanner = new Scanner(System.in);

		TankModel tm = new TankModel();
		tankview view = new tankview();

		System.out.println("***戦車検索***");
		while (true) {
			System.out.println("-----MENU------");

			scanner.nextLine();

			if (menu == 1) {
				ArrayList<Tank> tankList = tm.findAll();
				view.showlist(tankList);

			} else if (menu == 2) {
				ArrayList<Tank> tankList = tm.findAll();
				System.out.println("Enter Tank ID Number you want to know Details 👉");
				int index = scanner.nextInt();
				scanner.nextLine();
				if (index >= 1 && index <= tankList.size()) {
					Tank t = tm.findByAll(tankList.get(index - 1).getTankId());
					view.showDetail(t);
				} else {
					System.out.println("番号が範囲外です");
				}

			} else if (menu == 3) {
				System.out.println("キーワード入力してください");
				String word = scanner.nextLine();
				ArrayList<Tank> tankList = tm.findByKeyword(word);
				view.showlist(tankList);

			} else if (menu == 4) {
				System.out.println("年式の範囲を入力してください (最小 最大)");
				int yearMin = scanner.nextInt();
				int yearMax = scanner.nextInt();
				scanner.nextLine();

package team.view;

import java.io.FileWriter;
import java.io.IOException;
import java.nio.charset.StandardCharsets;
import java.util.ArrayList;

import team.entity.Tank;

public class tankhtmlview {

	public void showList(ArrayList<Tank> tankList) {
		try (FileWriter fw = new FileWriter("tankList.html", StandardCharsets.UTF_8)) {

			writeHeader(fw, "戦車リスト");

			if (tankList == null || tankList.isEmpty()) {
				fw.write("<p>該当する戦車がいません</p>");
			} else {
				fw.write("<table border=\"1\" cellpadding=\"8\">");
				fw.write("<tr><th>ID</th><th>画像</th><th>名前</th><th>国</th><th>年式</th></tr>");
				for (Tank t : tankList) {
					fw.write("<tr>");
					fw.write("<td>" + t.getTankId() + "</td>");
					fw.write("<td><img src=\"" + esc(t.getImage()) + "\" width=\"150\"></td>");
					fw.write("<td>" + esc(t.getName()) + "</td>");
					fw.write("<td>" + esc(t.getCountry()) + "</td>");
					fw.write("<td>" + t.getYear() + "</td>");
					fw.write("</tr>");
				}
				fw.write("</table>");
			}

			writeFooter(fw);
			System.out.println("tankList.html を作成しました");

		} catch (IOException e) {
			e.printStackTrace();
		}
	}

	public void showdetails(Tank tank) {
		if (tank == null) {
			System.out.println("戦車が見つかりません");
			return;
		}
		try (FileWriter fw = new FileWriter("tankDetail.html", StandardCharsets.UTF_8)) {

			writeHeader(fw, esc(tank.getName()));

			fw.write("<img src=\"" + esc(tank.getImage()) + "\" width=\"400\"><br><br>");
			fw.write("<table border=\"1\" cellpadding=\"8\">");
			fw.write("<tr><th>ID</th><td>" + tank.getTankId() + "</td></tr>");
			fw.write("<tr><th>名前</th><td>" + esc(tank.getName()) + "</td></tr>");
			fw.write("<tr><th>国</th><td>" + esc(tank.getCountry()) + "</td></tr>");
			fw.write("<tr><th>年式</th><td>" + tank.getYear() + "</td></tr>");
			fw.write("<tr><th>重量</th><td>" + tank.getWaight() + "kg</td></tr>");
			fw.write("<tr><th>口径</th><td>" + tank.getCaliber() + "mm</td></tr>");
			fw.write("<tr><th>種類</th><td>" + esc(tank.getTypes()) + "</td></tr>");
			fw.write("<tr><th>詳細</th><td>" + esc(tank.getDetails()) + "</td></tr>");
			fw.write("</table>");

			writeFooter(fw);
			System.out.println("tankDetail.html を作成しました");

		} catch (IOException e) {
			e.printStackTrace();
		}
	}

	private void writeHeader(FileWriter fw, String title) throws IOException {
		fw.write("<!DOCTYPE html>");
		fw.write("<html lang=\"ja\"><head>");
		fw.write("<meta charset=\"UTF-8\">");
		fw.write("<title>" + title + "</title>");
		fw.write("</head><body>");
		fw.write("<h1>" + title + "</h1>");
	}

	private void writeFooter(FileWriter fw) throws IOException {
		fw.write("</body></html>");
	}

	private String esc(String s) {
		if (s == null) {
			return "";
		}
		return s.replace("&", "&amp;")
				.replace("<", "&lt;")
				.replace(">", "&gt;")
				.replace("\"", "&quot;");
	}
}

パソコン売買システム

パソコンを売り買いできるコンソールアプリ。一覧・詳細・キーワード検索・価格検索に加えて、購入すると在庫が減り、出品すると新しい商品がデータベースに登録されます。在庫が足りないときは購入できないようにSQLでチェックしています。

package computer.controller;

import computer.entity.computer;
import computer.model.computermodel;
import computer.view.computerview;

public class computercontroller {

	public static void main(String[] args) {
		computermodel model = new computermodel();
		computerview view = new computerview();

		model.init(); // database04.db にテーブルが無ければ作成

		boolean loop = true;
		while (loop) {
			view.showMenu();
			int menu = view.readMenu();

			switch (menu) {
			case 1: // 一覧
				view.showList(model.findAll());
				break;

			case 2: { // 詳細
				System.out.print("IDを入力: ");
				int id = view.readInt();
				view.showDetail(model.findByAll(id));
				break;
			}

			case 3: { // キーワード検索
				System.out.print("キーワードを入力: ");
				String word = view.readString();
				view.showList(model.findByKeyword(word));
				break;
			}

			case 4: { // 価格検索
				System.out.print("価格(最小)を入力: ");
				long min = view.readLong();
				System.out.print("価格(最大)を入力: ");
				long max = view.readLong();
				view.showList(model.findbyprice(min, max));
				break;
			}

			case 5: { // 購入
				System.out.print("購入したいパソコンのIDを入力: ");
				int buyId = view.readInt();
				System.out.print("購入数量を入力: ");
				int qty = view.readInt();
				boolean ok = model.buyComputer(buyId, qty);
				view.showMessage(ok ? "購入が完了しました。" : "在庫不足、またはIDが存在しません。");
				break;
			}

			case 6: { // 出品
				System.out.print("商品名: ");
				String name = view.readString();
				System.out.print("ブランド: ");
				String brand = view.readString();
				System.out.print("CPU: ");
				String cpu = view.readString();
				System.out.print("GPU: ");
				String gpu = view.readString();
				System.out.print("RAM(GB): ");
				int ram = view.readInt();
				System.out.print("ストレージ(GB): ");
				int storage = view.readInt();
				System.out.print("カテゴリ: ");
				String category = view.readString();
				System.out.print("説明: ");
				String description = view.readString();
				System.out.print("画像URL: ");
				String image = view.readString();
				System.out.print("価格: ");
				long price = view.readLong();
				System.out.print("在庫数: ");
				int quantity = view.readInt();
				System.out.print("出品者名: ");
				String seller = view.readString();

				computer c = new computer(0, name, brand, cpu, gpu, ram, storage,
						category, description, image, price, quantity, seller);
				boolean ok = model.sellComputer(c);
				view.showMessage(ok ? "出品が完了しました。" : "出品に失敗しました。");
				break;
			}

			case 0:
				loop = false;
				view.showMessage("終了します。");
				break;

			default:
				view.showMessage("無効な選択です。");
			}
		}
	}
}
package computer.model;

import java.sql.Connection;
import java.sql.DriverManager;
import java.sql.PreparedStatement;
import java.sql.ResultSet;
import java.sql.SQLException;
import java.util.ArrayList;
import java.util.List;

import computer.entity.computer;

public class computermodel {

	private static final String URL = "jdbc:sqlite:computer.db";

	public void init() {
		String sql = "CREATE TABLE IF NOT EXISTS computer ("
				+ "computerId INTEGER PRIMARY KEY AUTOINCREMENT,"
				+ "name TEXT NOT NULL,"
				+ "brand TEXT,"
				+ "cpu TEXT,"
				+ "gpu TEXT,"
				+ "ram INTEGER,"
				+ "storage INTEGER,"
				+ "category TEXT,"
				+ "description TEXT,"
				+ "image TEXT,"
				+ "price INTEGER,"
				+ "quantity INTEGER,"
				+ "seller TEXT)";
		try (Connection Y = DriverManager.getConnection(URL);
				PreparedStatement X = Y.prepareStatement(sql)) {
			X.executeUpdate();
		} catch (SQLException e) {
			e.printStackTrace();
		}
	}

	public List<computer> findAll() {
		List<computer> list = new ArrayList<>();
		String sql = "SELECT * FROM computer ORDER BY computerId";
		try (Connection Y = DriverManager.getConnection(URL);
				PreparedStatement X = Y.prepareStatement(sql);
				ResultSet P = X.executeQuery()) {
			while (P.next()) {
				list.add(mapRow(P));
			}
		} catch (SQLException e) {
			e.printStackTrace();
		}
		return list;
	}

	public computer findByAll(int id) {
		computer c = null;
		String sql = "SELECT * FROM computer WHERE computerId = ?";
		try (Connection Y = DriverManager.getConnection(URL);
				PreparedStatement X = Y.prepareStatement(sql)) {
			X.setInt(1, id);
			try (ResultSet P = X.executeQuery()) {
				if (P.next()) {
					c = mapRow(P);
				}
			}
		} catch (SQLException e) {
			e.printStackTrace();
		}
		return c;
	}

	public List<computer> findByKeyword(String word) {
		List<computer> list = new ArrayList<>();
		String sql = "SELECT * FROM computer WHERE name LIKE ? OR brand LIKE ? OR category LIKE ?";
		String w = "%" + word + "%";
		try (Connection Y = DriverManager.getConnection(URL);
				PreparedStatement X = Y.prepareStatement(sql)) {
			X.setString(1, w);
			X.setString(2, w);
			X.setString(3, w);
			try (ResultSet P = X.executeQuery()) {
				while (P.next()) {
					list.add(mapRow(P));
				}
			}
		} catch (SQLException e) {
			e.printStackTrace();
		}
		return list;
	}

	public List<computer> findbyprice(long min, long max) {
		List<computer> list = new ArrayList<>();
		String sql = "SELECT * FROM computer WHERE price BETWEEN ? AND ? ORDER BY price";
		try (Connection Y = DriverManager.getConnection(URL);
				PreparedStatement X = Y.prepareStatement(sql)) {
			X.setLong(1, min);
			X.setLong(2, max);
			try (ResultSet P = X.executeQuery()) {
				while (P.next()) {
					list.add(mapRow(P));
				}
			}
		} catch (SQLException e) {
			e.printStackTrace();
		}
		return list;
	}

	public boolean buyComputer(int id, int qty) {
		String sql = "UPDATE computer SET quantity = quantity - ? WHERE computerId = ? AND quantity >= ?";
		try (Connection Y = DriverManager.getConnection(URL);
				PreparedStatement X = Y.prepareStatement(sql)) {
			X.setInt(1, qty);
			X.setInt(2, id);
			X.setInt(3, qty);
			return X.executeUpdate() > 0;
		} catch (SQLException e) {
			e.printStackTrace();
			return false;
		}
	}

	public boolean sellComputer(computer c) {
		String sql = "INSERT INTO computer (name,brand,cpu,gpu,ram,storage,category,description,image,price,quantity,seller) "
				+ "VALUES (?,?,?,?,?,?,?,?,?,?,?,?)";
		try (Connection Y = DriverManager.getConnection(URL);
				PreparedStatement X = Y.prepareStatement(sql)) {
			X.setString(1, c.getName());
			X.setString(2, c.getBrand());
			X.setString(3, c.getCpu());
			X.setString(4, c.getGpu());
			X.setInt(5, c.getRam());
			X.setInt(6, c.getStorage());
			X.setString(7, c.getCategory());
			X.setString(8, c.getDescription());
			X.setString(9, c.getImage());
			X.setLong(10, c.getPrice());
			X.setInt(11, c.getQuantity());
			X.setString(12, c.getSeller());
			return X.executeUpdate() > 0;
		} catch (SQLException e) {
			e.printStackTrace();
			return false;
		}
	}

	public boolean update(computer c) {
		String sql = "UPDATE computer SET name=?,brand=?,cpu=?,gpu=?,ram=?,storage=?,category=?,description=?,image=?,price=?,quantity=?,seller=? WHERE computerId=?";
		try (Connection Y = DriverManager.getConnection(URL);
				PreparedStatement X = Y.prepareStatement(sql)) {
			X.setString(1, c.getName());
			X.setString(2, c.getBrand());
			X.setString(3, c.getCpu());

import java.io.FileInputStream;
import java.util.Random;
import java.util.Scanner;

import javazoom.jl.player.Player;

public class Project1 {

	static final int MAX_HP = 100;
	static final int MAX_HEAL = 3;

	public static void main(String[] args) {
		Scanner sc = new Scanner(System.in);
		Random r = new Random();

		int[] hp = { MAX_HP, MAX_HP };
		boolean[] defending = { false, false };
		int[] healcount = { 0, 0 };
		String[] p = new String[2];

		int current = 0;

		// Player names
		System.out.print("Enter Player 1 Name: ");
		p[0] = sc.nextLine();
		playSound("rizz-sound-effect.mp3");

		System.out.print("Enter Player 2 Name: ");
		p[1] = sc.nextLine();
		playSound("rizz-sound-effect.mp3");

		int total = getBet(sc, p[0]) + getBet(sc, p[1]);

		while (hp[0] > 0 && hp[1] > 0) {
			int opponent = 1 - current;

			displayStatus(p, hp, defending, healcount);

			System.out.println("\n" + p[current] + "'s Turn");
			System.out.println("1. Attack  2. Defend  3. Heal");

			int choice = sc.nextInt();

			switch (choice) {
			case 1:
				attack(r, p, hp, defending, current, opponent);
				break;

			case 2:
				defending[current] = true;
				System.out.println(p[current] + " is defending!");
				break;

			case 3:
				heal(r, p, hp, healcount, current);
				break;

			default:
				System.out.println("Invalid choice!");
				playSound("vine-boom.mp3");
				continue;
			}

			if (hp[opponent] <= 0)
				break;

			current = opponent;
		}

		System.out.println("\n===== GAME OVER =====");
		if (hp[0] <= 0) {
			System.out.println(p[1] + " Wins!");
		} else {
			System.out.println(p[0] + " Wins!");
		}
		playSound("rizz-sound-effect.mp3");
		System.out.println("CONGRALUTIONS YOU WIN ￥" + total);

	}

	static void attack(Random r, String[] p, int[] hp, boolean[] defending, int current, int opponent) {
		int damage = r.nextInt(11) + 10;
		if ((r.nextInt(2) == 0)) {
			playSound("重いキック1.mp3");
		} else {
			playSound("punch-gaming-sound-effect-hd_RzlG1GE.mp3");
		}

		if (r.nextInt(100) < 20) {
			damage *= 2;
			System.out.println("🔥 CRITICAL HIT!");
			playSound("bone-crack.mp3");
		}

		if (defending[opponent]) {
			damage /= 2;
			defending[opponent] = false;
			playSound("rizz-sound-effect.mp3");
		}

		hp[opponent] -= damage;
		System.out.println(p[current] + " attacked for " + damage + " damage!");
	}

	static void heal(Random r, String[] p, int[] hp, int[] healcount, int current) {
		if (healcount[current] >= MAX_HEAL) {
			System.out.println("❌ No heals left!");
			playSound("fahhhhhhhhhhhhhh.mp3");
			return;
		}

		int heal = r.nextInt(20) + 5;
		hp[current] = Math.min(MAX_HP, hp[current] + heal);
		healcount[current]++;

		playSound("anime-wow-sound-effect.mp3");
		System.out.println(p[current] + " healed " + heal + " HP!");
	}

	static int getBet(Scanner sc, String player) {
		System.out.print(player + ", enter your bet: ");
		int bet = sc.nextInt();

		if (bet < 1000) {
			playSound("999-social-credit-siren.mp3");
		} else {
			playSound("anime-wow-sound-effect.mp3");
		}

		return bet;
	}

	static void displayStatus(String[] p, int[] hp, boolean[] defending, int[] healcount) {
		for (int i = 0; i < 2; i++) {
			System.out.println(p[i] + " HP: " + hp[i] +
					(defending[i] ? " (Defending)" : "") +
					" | Heals left: " + (MAX_HEAL - healcount[i]));
		}
	}

	static void playSound(String fileName) {
		try {
			FileInputStream file = new FileInputStream(fileName);
			Player sound = new Player(file);
			sound.play();
		} catch (Exception e) {
			System.out.println("⚠ Sound error: " + fileName);
		}
	}
}
